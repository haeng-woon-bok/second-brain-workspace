function parseAndSetNewsData(rawNews) { const parseSection = (text, sectorName) => { if (!text) return []; // 항목 단위 분할 (• 제목: 또는 단순 줄바꿈 대응) const rawItems = text.split(/(?=•\s*제목:)/).filter(b => b.trim()); return rawItems.map((raw, idx) => { const titleMatch = raw.match(/•\s*제목:\s*([^\n]+)/); const linkMatch = raw.match(/링크:\s*([^\n]+)/); const summaryMatch = raw.match(/요약:\s*([\s\S]+)/); const title = titleMatch ? titleMatch[1].trim() : (raw.split('\n')[0] || '뉴스 제목'); const link = linkMatch ? linkMatch[1].trim() : '#'; const summary = summaryMatch ? summaryMatch[1].trim().slice(0, 150) : raw.slice(0, 100); return { id: `${sectorName}_${idx}`, sector: sectorName, title: title, link: link, summary: summary, keywords: [sectorName, '실시간'] }; }); }; NEWS_DATA = { stock: parseSection(rawNews['증시'], '증시/매크로'), crypto: parseSection(rawNews['비트코인/이더리움'], '가상자산'), realestate: parseSection(rawNews['부동산'], '부동산/정비') }; }
